@dataclass(frozen=True)
class V51Evidence:
    evaluated_rows: tuple[dict[str, object], ...]
    baseline_by_game: dict[int, str]
    summary: dict[str, object]
    lock_audit: dict[str, object]
    model_card: dict[str, object]
    predictions_content_sha256: str
    evaluated_content_sha256: str

def exact_mcnemar(model_only: int, baseline_only: int) -> float:
    discordant = model_only + baseline_only
    if discordant == 0:
        return 1.0
    tail = sum(math.comb(discordant, k) for k in range(min(model_only, baseline_only) + 1)) / (2 ** discordant)
    return min(1.0, 2.0 * tail)

def paired_delta_interval(model_only: int, baseline_only: int, n: int, z: float = 1.96) -> tuple[float, float]:
    delta = (model_only - baseline_only) / n
    variance = ((model_only + baseline_only) / n - delta * delta) / n
    margin = z * math.sqrt(variance)
    return delta - margin, delta + margin

def canonical_json_bytes(value: object) -> bytes:
    return json.dumps(value, allow_nan=False, ensure_ascii=False, separators=(",", ":"), sort_keys=True).encode("utf-8")

def sha256_file(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()
```

def execute_notebook_cells(sources: Sequence[str], globals_dict: dict[str, object]) -> list[dict[str, object]]:
    cells = []
    for count, source in enumerate(sources, 1):
        stream = io.StringIO()
        with contextlib.redirect_stdout(stream):
            exec(compile(source, f"diagnostic_100_cell_{count}", "exec"), globals_dict)
        cells.append({
            "cell_type": "code", "execution_count": count, "metadata": {}, "source": source.splitlines(keepends=True),
            "outputs": [{"name": "stdout", "output_type": "stream", "text": stream.getvalue().splitlines(keepends=True)}],
        })
    return cells
```

def classify_v52(probability: float, completeness: float, uncertainty: str, *, critical: bool) -> str:
